cmake_minimum_required(VERSION 3.20)
project(breathon_preview C)
set(CMAKE_C_STANDARD 99)
option(BREATHON_UI_REDUCED_MOTION "Render static decorative effects" OFF)
set(LVGL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/lvgl-native" CACHE PATH "LVGL 9.1 checkout")
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_DIR}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_include_directories(lvgl PUBLIC "${LVGL_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/sim")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
file(GLOB UI_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c")
add_library(breathon_ui STATIC ${UI_SOURCES})
target_include_directories(breathon_ui PUBLIC include)
target_link_libraries(breathon_ui PUBLIC lvgl)
target_compile_options(breathon_ui PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(breathon_ui PUBLIC BREATHON_UI_REDUCED_MOTION=$<BOOL:${BREATHON_UI_REDUCED_MOTION}>)
add_executable(breathon_preview sim/main.c sim/capture.c sim/audio_tests.c sim/voice_tests.c sim/motion_tests.c sim/showcase_tests.c
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/font_puhui_18_2.c")
target_link_libraries(breathon_preview PRIVATE breathon_ui m)
target_compile_options(breathon_preview PRIVATE -Wall -Wextra -Werror -UNDEBUG)
enable_testing()
add_executable(timeline_tests sim/timeline_tests.c src/ui_timeline.c)
target_compile_options(timeline_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME timeline COMMAND timeline_tests)
add_executable(request_tests sim/request_tests.c ../firmware/realtime_request.c)
target_compile_options(request_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME request_lifecycle COMMAND request_tests)
add_test(NAME native_preview COMMAND breathon_preview)
set_tests_properties(native_preview PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
